  function gcd(a, b) {
    a = Math.abs(a);
    b = Math.abs(b);
    while (b !== 0) {
      const t = b;
      b = a % b;
      a = t;
    }
    return a || 1;
  }

  function renderVerticalFraction(num, den) {
    return `<span class="v-frac"><span class="v-frac-num">${num}</span><span class="v-frac-den">${den}</span></span>`;
  }

  function formatLinearFactor(coefX, constant) {
    const xPart = coefX === 1 ? 'x' : `${coefX}x`;
    if (constant === 0) return `(${xPart})`;
    return `(${xPart} ${constant >= 0 ? '+' : '-'} ${Math.abs(constant)})`;
  }

  function formatQuadratic(a, b, c) {
    let str = a === 1 ? 'x²' : `${a}x²`;
    if (b !== 0) {
      const absB = Math.abs(b);
      str += b > 0 ? ` + ${absB === 1 ? '' : absB}x` : ` - ${absB === 1 ? '' : absB}x`;
    }
    if (c !== 0) {
      str += c > 0 ? ` + ${c}` : ` - ${Math.abs(c)}`;
    }
    return str;
  }

  function generateQuadraticProblem(difficultyMode, currentWave) {
    let effectiveDiff = difficultyMode;
    if (difficultyMode === 'intense') {
      if (currentWave <= 3) effectiveDiff = 'easy';
      else if (currentWave <= 7) effectiveDiff = 'medium';
      else effectiveDiff = 'hard';
    }

    const isSolve = Math.random() < 0.25; // 75% Factoring, 25% Solving
    let a = 1, p = 1, q = 1, r = 1, s = 1;
    let attempts = 0;

    while (attempts < 90) {
      attempts++;
      if (effectiveDiff === 'easy') {
        a = 1; p = 1; r = 1;
        q = (Math.floor(Math.random() * 11) + 1) * (Math.random() < 0.5 ? 1 : -1);
        s = (Math.floor(Math.random() * 11) + 1) * (Math.random() < 0.5 ? 1 : -1);
        if (Math.abs(q * s) < 70 && (q + s) !== 0) break;
      } else if (effectiveDiff === 'medium') {
        const aPool = [1, 2, 2, 3, 4];
        a = aPool[Math.floor(Math.random() * aPool.length)];
        if (a === 1) { p = 1; r = 1; }
        else if (a === 2) { p = 2; r = 1; }
        else if (a === 3) { p = 3; r = 1; }
        else { p = 2; r = 2; }
        q = (Math.floor(Math.random() * 8) + 1) * (Math.random() < 0.5 ? 1 : -1);
        s = (Math.floor(Math.random() * 8) + 1) * (Math.random() < 0.5 ? 1 : -1);
        if (gcd(p, q) === 1 && gcd(r, s) === 1) {
          const cVal = q * s;
          if (Math.abs(a * cVal) < 100 && (p * s + q * r) !== 0) break;
        }
      } else {
        const pairs = [[2,1],[3,1],[3,2],[4,1],[4,3],[5,1],[5,2],[6,1]];
        const chosen = pairs[Math.floor(Math.random() * pairs.length)];
        p = chosen[0];
        r = chosen[1];
        a = p * r;
        q = (Math.floor(Math.random() * 9) + 1) * (Math.random() < 0.5 ? 1 : -1);
        s = (Math.floor(Math.random() * 9) + 1) * (Math.random() < 0.5 ? 1 : -1);
        if (gcd(p, q) === 1 && gcd(r, s) === 1) {
          const cVal = q * s;
          if (Math.abs(a * cVal) < 200 && (p * s + q * r) !== 0) break;
        }
      }
    }

    a = p * r;
    const b = p * s + q * r;
    const c = q * s;
    const ac = a * c;

    const m1 = q * r;
    const m2 = p * s;

    const g1 = gcd(m1, a);
    const simp1Num = m1 / g1;
    const simp1Den = a / g1;

    const g2 = gcd(m2, a);
    const simp2Num = m2 / g2;
    const simp2Den = a / g2;

    const polyStr = formatQuadratic(a, b, c);
    const equationStr = isSolve ? `${polyStr} = 0` : polyStr;
    const taskType = isSolve ? 'SOLVE FOR X:' : 'FACTOR COMPLETELY:';

    let correctAnswerHtml = '';
    let distractors = [];

    if (!isSolve) {
      const f1 = formatLinearFactor(simp1Den, simp1Num);
      const f2 = formatLinearFactor(simp2Den, simp2Num);
      correctAnswerHtml = f1 === f2 ? `${f1}²` : `${f1}${f2}`;

      const d1f1 = formatLinearFactor(simp1Den, -simp1Num);
      const d1f2 = formatLinearFactor(simp2Den, -simp2Num);
      const d1 = d1f1 === d1f2 ? `${d1f1}²` : `${d1f1}${d1f2}`;
      const d2 = `${formatLinearFactor(1, m1)}${formatLinearFactor(1, m2)}`;
      const d3 = `${formatLinearFactor(simp1Den, -simp1Num)}${formatLinearFactor(simp2Den, simp2Num)}`;

      const vFrac1 = renderVerticalFraction(simp1Num, simp1Den);
      const vFrac2 = renderVerticalFraction(simp2Num, simp2Den);

      distractors = [
        { html: d1, reason: 'Sign Error: Both signs are reversed! Your X-Method numbers m₁ and m₂ must multiply to A·C AND add to B.' },
        { html: d2, reason: a > 1 ? `Forgot to Divide by a = ${a}: You found m₁ = ${m1} and m₂ = ${m2}, but forgot to place them over a = ${a} and simplify (${vFrac1} and ${vFrac2}).` : 'Factor Pair Error: Double-check the two integers that multiply to A·C and add to B.' },
        { html: d3, reason: 'Partial Sign Error: One binomial has the wrong sign. Multiply your factors back out using FOIL to check the middle term B.' }
      ];
    } else {
      const fmtRootHtml = (num, den) => {
        const rootNum = -num;
        return den === 1 ? `${rootNum}` : renderVerticalFraction(rootNum, den);
      };
      const r1 = fmtRootHtml(simp1Num, simp1Den);
      const r2 = fmtRootHtml(simp2Num, simp2Den);
      correctAnswerHtml = r1 === r2 ? `x = ${r1}` : `x = ${r1}, &nbsp;x = ${r2}`;

      const w1r1 = fmtRootHtml(-simp1Num, simp1Den);
      const w1r2 = fmtRootHtml(-simp2Num, simp2Den);
      const d1 = `x = ${w1r1}, &nbsp;x = ${w1r2}`;
      const d2 = `x = ${-m1}, &nbsp;x = ${-m2}`;
      const d3 = `x = ${w1r1}, &nbsp;x = ${r2}`;

      const vFrac1 = renderVerticalFraction(simp1Num, simp1Den);
      const vFrac2 = renderVerticalFraction(simp2Num, simp2Den);

      distractors = [
        { html: d1, reason: `Zero-Product Sign Error: From the X-Method fractions (${vFrac1} and ${vFrac2}), setting each factor equal to 0 flips the signs to solve for x!` },
        { html: d2, reason: a > 1 ? `Forgot to Divide by a = ${a}: After finding m₁ = ${m1} and m₂ = ${m2}, divide both by a = ${a} before solving.` : 'Pair Selection Error: Re-check which two numbers multiply to A·C and add to B.' },
        { html: d3, reason: `Sign Flip Error on First Root: Remember (px + q) = 0 solves to x = ${renderVerticalFraction('-q', 'p')}.` }
      ];
    }

    const used = new Set([correctAnswerHtml]);
    const finalOptions = [{ html: correctAnswerHtml, isCorrect: true, reason: '' }];
    for (const d of distractors) {
      if (!used.has(d.html)) {
        used.add(d.html);
        finalOptions.push({ html: d.html, isCorrect: false, reason: d.reason });
      }
    }
    let fallbackOffset = 1;
    while (finalOptions.length < 4) {
      const alt = !isSolve
        ? `${formatLinearFactor(simp1Den, simp1Num + fallbackOffset)}${formatLinearFactor(simp2Den, simp2Num - fallbackOffset)}`
        : `x = ${-simp1Num + fallbackOffset}, &nbsp;x = ${-simp2Num}`;
      fallbackOffset++;
      if (!used.has(alt)) {
        used.add(alt);
        finalOptions.push({ html: alt, isCorrect: false, reason: 'Arithmetic Error: Those values do not satisfy both A·C and B in the X-Method diamond.' });
      }
    }

    for (let i = finalOptions.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [finalOptions[i], finalOptions[j]] = [finalOptions[j], finalOptions[i]];
    }

    const step1 = `Step 1 (Diamond Setup): Top A·C = (${a})(${c}) = ${ac}, Bottom B = ${b}.`;
    const step2 = `Step 2 (Find Wings): Two numbers multiplying to ${ac} and adding to ${b} are m₁ = ${m1} and m₂ = ${m2}.`;
    const step3 = a > 1
      ? `Step 3 (Divide by a = ${a} & Simplify): ${renderVerticalFraction(m1, a)} → ${renderVerticalFraction(simp1Num, simp1Den)} and ${renderVerticalFraction(m2, a)} → ${renderVerticalFraction(simp2Num, simp2Den)}. Result: ${correctAnswerHtml}.`
      : `Step 3 (Write Result): Since a = 1, read directly from ${renderVerticalFraction(m1, 1)} and ${renderVerticalFraction(m2, 1)} → ${correctAnswerHtml}.`;

    return {
      a, b, c, ac, m1, m2,
      simp1Num, simp1Den, simp2Num, simp2Den,
      isSolve, taskType, equationStr, correctAnswerHtml,
      options: finalOptions,
      steps: [step1, step2, step3]
    };
  }

  function recalculateMaxHp(p) {
    const integrityFactor = Math.max(0.20, 1 - p.deaths * 0.20);
    p.maxHp = Math.max(20, Math.round((p.baseMaxHp + p.bonusMaxHp) * integrityFactor));
    if (p.hp > p.maxHp) p.hp = p.maxHp;
  }

  function handlePlayerDamage(rawDmg) {
    const p = STATE.player;
    if (p.invulnTimer > 0) return;

    const taken = Math.max(2, rawDmg * (1 - p.armor));
    p.hp -= taken;
    p.invulnTimer = 0.52;
    STATE.cameraShake = 8;
    spawnFloatingText(p.x, p.y - 26, `-${Math.round(taken)}`, '#fb7185', 15);

    if (p.hp <= 0) {
      p.deaths += 1;
      if (p.deaths >= p.maxDeaths) {
        p.hp = 0;
        updateHUD();
        endGame(false);
        return;
      }
      // Emergency Phase-Revive: -20% Max HP Integrity Penalty & Knockback Shockwave
      recalculateMaxHp(p);
      p.hp = p.maxHp;
      p.invulnTimer = 2.2;
      STATE.cameraShake = 16;
      sfx.play('revive');

      STATE.shockwaves.push({ x: p.x, y: p.y, radius: 18, maxRadius: 340, color: '#fbbf24', alpha: 1.0 });
      STATE.enemies.forEach(e => {
        const ang = Math.atan2(e.y - p.y, e.x - p.x);
        if (!e.isBoss) {
          e.x += Math.cos(ang) * 140;
          e.y += Math.sin(ang) * 140;
        }
        e.stunTimer = Math.max(e.stunTimer, 2.5);
      });
      spawnFloatingText(p.x, p.y - 44, `SUIT INTEGRITY -20% (${p.maxDeaths - p.deaths} LIVES LEFT)`, '#fbbf24', 18);
    }
  }

  function addOrMergeWeapon(weaponId) {
    STATE.weapons.push({ id: weaponId, tier: 1, cooldown: 0 });
    let merged = true;
    while (merged) {
      merged = false;
      for (let t = 1; t <= 3; t++) {
        const counts = {};
        STATE.weapons.forEach(w => {
          if (w.tier === t) counts[w.id] = (counts[w.id] || 0) + 1;
        });
        for (const id in counts) {
          if (counts[id] >= 2) {
            let removed = 0;
            STATE.weapons = STATE.weapons.filter(w => {
              if (w.id === id && w.tier === t && removed < 2) {
                removed++;
                return false;
              }
              return true;
            });
            const nextTier = t + 1;
            STATE.weapons.push({ id, tier: nextTier, cooldown: 0 });
            const tierStyle = TIER_STYLES[nextTier];
            spawnFloatingText(
              STATE.player.x,
              STATE.player.y - 40,
              nextTier === 4 ? `★ FORGED TIER 4 SOVEREIGN GOLD! ★` : `UPGRADED TO ${tierStyle.label}!`,
              tierStyle.color,
              nextTier === 4 ? 20 : 17
            );
            sfx.play('levelup');
            merged = true;
            break;
          }
        }
        if (merged) break;
      }
    }
    if (STATE.weapons.length > 6) {
      STATE.weapons = STATE.weapons.slice(0, 6);
    }
    updateHUDWeapons();
  }

  function triggerDash() {
    if (STATE.mode !== 'PLAYING') return;
    const p = STATE.player;
    if (p.dashCharges <= 0 || p.dashActiveTimer > 0) return;

    p.dashCharges -= 1;
    p.dashActiveTimer = 0.18;
    p.invulnTimer = 0.30;

    let dx = 0, dy = 0;
    if (STATE.keys['KeyW'] || STATE.keys['ArrowUp']) dy -= 1;
    if (STATE.keys['KeyS'] || STATE.keys['ArrowDown']) dy += 1;
    if (STATE.keys['KeyA'] || STATE.keys['ArrowLeft']) dx -= 1;
    if (STATE.keys['KeyD'] || STATE.keys['ArrowRight']) dx += 1;
    if (STATE.joystick.active) {
      dx = STATE.joystick.dx;
      dy = STATE.joystick.dy;
    }
    const len = Math.hypot(dx, dy);
    if (len > 0.05) {
      p.dashDirX = dx / len;
      p.dashDirY = dy / len;
    } else {
      p.dashDirX = Math.cos(p.facingAngle);
      p.dashDirY = Math.sin(p.facingAngle);
    }

    sfx.play('dash');
    STATE.shockwaves.push({ x: p.x, y: p.y, radius: 10, maxRadius: 80, color: '#38bdf8', alpha: 0.9 });

    if (p.dashNovaLevel > 0) {
      const novaRadius = 135 + p.dashNovaLevel * 28;
      const novaDmg = (52 + p.dashNovaLevel * 32) * p.damageMult;
      STATE.shockwaves.push({ x: p.x, y: p.y, radius: 15, maxRadius: novaRadius, color: '#fbbf24', alpha: 1.0 });
      STATE.enemies.forEach(e => {
        if (Math.hypot(e.x - p.x, e.y - p.y) <= novaRadius + e.radius) {
          damageEnemy(e, novaDmg, true);
          e.stunTimer = Math.max(e.stunTimer, 1.2);
        }
      });
    }
  }

  function triggerUltimate() {
    if (STATE.mode !== 'PLAYING') return;
    const p = STATE.player;
    if (p.ultCharge < 100) return;
    p.ultCharge = 0;
    STATE.cameraShake = 12;
    sfx.play('ult');

    spawnFloatingText(p.x, p.y - 42, `${p.hero.name.toUpperCase()} ULTIMATE!`, '#fbbf24', 20);

    if (id === 'lyra' || id === 'terra') {
      const heal = 65;
      p.hp = Math.min(p.maxHp, p.hp + heal);
      spawnFloatingText(p.x, p.y - 22, `+${heal} HP`, '#34d399', 18);

    if (id === 'ada') {
      p.overclockTimer = 5.0;
    }
    if (id === 'rook' || id === 'glacia' || id === 'euclid') {
      STATE.enemies.forEach(e => {
        e.stunTimer = 4.0;
        e.chillTimer = 5.0;
        damageEnemy(e, 75 * p.damageMult, false);
      });
      if (id === 'euclid') {
        STATE.gems.forEach(g => { g.magnetized = true; });
      }
      return;
    }

    const count = 24;
    for (let i = 0; i < count; i++) {
      const ang = (Math.PI * 2 * i) / count;
      STATE.projectiles.push({
        x: p.x,
        y: p.y,

  function damageEnemy(enemy, amount, isCrit = false) {
    enemy.hp -= amount;
    enemy.hitFlash = 0.08;
    if (isCrit) {
      spawnFloatingText(enemy.x, enemy.y - enemy.radius - 6, `${Math.round(amount)}!`, '#fbbf24', 15);
    }
    if (enemy.hp <= 0 && !enemy.dead) {
      enemy.dead = true;
      STATE.stats.kills += 1;
      const p = STATE.player;
      p.ultCharge = Math.min(100, p.ultCharge + (enemy.isBoss ? 50 : 3.8));
      if (p.lifeSteal > 0) {
        p.hp = Math.min(p.maxHp, p.hp + p.lifeSteal);
      }

      STATE.gems.push({
        x: enemy.x,
        y: enemy.y,
        xp: Math.round(enemy.xp * p.xpMult),
        gold: Math.round(enemy.gold * p.xpMult),
        isBoss: enemy.isBoss || enemy.alienLevel >= 7,
        magnetized: false
      });

      const burstCount = enemy.isBoss ? 36 : (6 + enemy.alienLevel);
      for (let i = 0; i < burstCount; i++) {
        const a = Math.random() * Math.PI * 2;
        const spd = 65 + Math.random() * 210;
        STATE.particles.push({
          x: enemy.x,
          y: enemy.y,
          vx: Math.cos(a) * spd,
          vy: Math.sin(a) * spd,
          life: 0.35 + Math.random() * 0.25,
          maxLife: 0.6,
          size: 3 + Math.random() * 4.5,
          color: i % 2 === 0 ? enemy.color1 : enemy.glow
        });
      }
      sfx.play('kill');
    }
  }

  // =========================================================================
  // 17. MAIN SIMULATION STEP
  // =========================================================================
  function update(dt) {
    if (STATE.mode !== 'PLAYING') return;

    STATE.time += dt;
    STATE.waveTimer -= dt;
    if (STATE.cameraShake > 0) STATE.cameraShake = Math.max(0, STATE.cameraShake - dt * 28);

    const p = STATE.player;

    for (const mark of STATE.breachTriggers) {
      if (STATE.waveTimer <= mark && !STATE.triggeredBreaches.has(mark)) {
        STATE.triggeredBreaches.add(mark);
        const mins = Math.floor(mark / 60);
        const secs = mark % 60;
        openQuadraticBreach(`TIMED BREACH (${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')})`, false);
        return;
      }
    }

    if (STATE.waveTimer <= 0) {
      STATE.waveTimer = 0;
      updateHUD();
      openQuadraticBreach(`WAVE ${STATE.wave} EXTRACTION BREACH`, true);
      return;
    }

    if (p.invulnTimer > 0) p.invulnTimer -= dt;
    if (p.overclockTimer > 0) p.overclockTimer -= dt;
    if (p.recoil > 0) p.recoil = Math.max(0, p.recoil - dt * 26);
    p.orbitAngle += dt * 3.5;

    if (p.regen > 0 && p.hp < p.maxHp) {
      p.hp = Math.min(p.maxHp, p.hp + p.regen * dt);
    }

    if (p.dashCharges < p.maxDashes) {
      p.dashRechargeTimer += dt;
      if (p.dashRechargeTimer >= p.dashCooldownBase) {
        p.dashRechargeTimer = 0;
        p.dashCharges += 1;
      }
    }

    let moveX = 0, moveY = 0;
    p.isMoving = false;
    if (p.dashActiveTimer > 0) {
      p.dashActiveTimer -= dt;
      p.isMoving = true;
      const dashSpd = 800;
      p.x += p.dashDirX * dashSpd * dt;
      p.y += p.dashDirY * dashSpd * dt;
      STATE.particles.push({
        x: p.x, y: p.y, vx: 0, vy: 0,
        life: 0.18, maxLife: 0.18, size: 13, color: '#38bdf8'
      });
    } else {
      if (STATE.keys['KeyW'] || STATE.keys['ArrowUp']) moveY -= 1;
      if (STATE.keys['KeyS'] || STATE.keys['ArrowDown']) moveY += 1;
      if (STATE.keys['KeyA'] || STATE.keys['ArrowLeft']) moveX -= 1;
      if (STATE.keys['KeyD'] || STATE.keys['ArrowRight']) moveX += 1;
      if (STATE.joystick.active) {
        moveX = STATE.joystick.dx;
        moveY = STATE.joystick.dy;
      }
      const len = Math.hypot(moveX, moveY);
      if (len > 0.01) {
        p.isMoving = true;
        const nx = moveX / len;
        const ny = moveY / len;
        const spd = p.baseSpeed * p.speedMult;
        p.x += nx * spd * dt;
        p.y += ny * spd * dt;
        p.walkCycle += dt * 14;
      }
    }

    const margin = 28;
    p.x = Math.max(margin, Math.min(STATE.width - margin, p.x));
    p.y = Math.max(margin + 56, Math.min(STATE.height - margin, p.y));

    let nearestEnemy = null;
    let nearestDist = Infinity;
    for (const e of STATE.enemies) {
      const d = Math.hypot(e.x - p.x, e.y - p.y);
      if (d < nearestDist) {
        nearestDist = d;
        nearestEnemy = e;
      }
    }

    if (STATE.autoAim && nearestEnemy) {
      p.facingAngle = Math.atan2(nearestEnemy.y - p.y, nearestEnemy.x - p.x);
    } else {
      p.facingAngle = Math.atan2(STATE.mouse.y - p.y, STATE.mouse.x - p.x);
    }

    const effectiveRateMult = p.fireRateMult * (p.overclockTimer > 0 ? 2.5 : 1.0);
    for (const wSlot of STATE.weapons) {
      const meta = WEAPON_CATALOG[wSlot.id];
      const tStyle = TIER_STYLES[wSlot.tier] || TIER_STYLES[1];
      wSlot.cooldown -= dt * effectiveRateMult;
      if (wSlot.cooldown <= 0 && (STATE.enemies.length > 0 || !STATE.autoAim)) {
        wSlot.cooldown = meta.rate * tStyle.rateMult;
        p.recoil = 6.5;
        sfx.play('shoot');

        const totalCount = meta.count + p.bonusProjectiles + tStyle.extraCount;
        const baseAngle = p.facingAngle;

        for (let i = 0; i < totalCount; i++) {
          const offsetIndex = i - (totalCount - 1) / 2;
          const ang = baseAngle + offsetIndex * (meta.spread || 0.09) + (Math.random() - 0.5) * 0.03;
          const isCrit = Math.random() < p.critChance || wSlot.tier === 4;
          const dmg = meta.dmg * tStyle.dmgMult * p.damageMult * (isCrit ? p.critMult : 1.0);
          const spd = meta.speed * p.projSpeedMult * (wSlot.tier === 4 ? 1.25 : 1.0);

          STATE.projectiles.push({
            x: p.x + Math.cos(ang) * 24,
            y: p.y + Math.sin(ang) * 24,
            vx: Math.cos(ang) * spd,
            vy: Math.sin(ang) * spd,
            radius: wSlot.tier === 4 ? 8.5 : (4.5 + wSlot.tier * 0.8),
            dmg,
            pierce: meta.pierce + p.bonusPierce + tStyle.extraPierce,
            life: 1.55,
            color: tStyle.color,
            isGold: wSlot.tier === 4,
            crit: isCrit,
            chill: !!meta.chill,
            homing: !!meta.homing || wSlot.tier === 4,
            bounce: !!meta.bounce,
            explode: Math.max(meta.explode || 0, tStyle.goldExplode) * p.areaMult,
            hitSet: new Set()
          });
        }
      }
    }

    if (p.orbitShields > 0) {
      const orbitR = 58;
      for (let i = 0; i < p.orbitShields; i++) {
        const a = p.orbitAngle + (Math.PI * 2 * i) / p.orbitShields;
        const sx = p.x + Math.cos(a) * orbitR;
        const sy = p.y + Math.sin(a) * orbitR;
        for (const e of STATE.enemies) {
          if (Math.hypot(e.x - sx, e.y - sy) < e.radius + 14) {
            damageEnemy(e, 45 * p.damageMult * dt * 5, false);
          }
        }
      }
    }

    const icfg = getIntensityCfg();
    const targetActiveEnemies = Math.max(10, Math.round(Math.min(68, 18 + STATE.wave * 4) * icfg.countMult));
    const spawnInterval = Math.max(0.10, (0.72 - STATE.wave * 0.045) / icfg.rateMult);
    STATE.spawnAccumulator += dt;
    if (STATE.spawnAccumulator >= spawnInterval && STATE.enemies.length < targetActiveEnemies) {
      STATE.spawnAccumulator = 0;
      spawnAlienByLevel(pickAlienLevelForWave(STATE.wave));
    }

    for (let i = STATE.projectiles.length - 1; i >= 0; i--) {
      const proj = STATE.projectiles[i];
      proj.life -= dt;
      if (proj.life <= 0) {
        STATE.projectiles.splice(i, 1);
        continue;
      }

      if (proj.homing && nearestEnemy) {
        const desired = Math.atan2(nearestEnemy.y - proj.y, nearestEnemy.x - proj.x);
        const cur = Math.atan2(proj.vy, proj.vx);
        const spd = Math.hypot(proj.vx, proj.vy);
        const diff = Math.atan2(Math.sin(desired - cur), Math.cos(desired - cur));
        const next = cur + diff * Math.min(1, dt * (proj.isGold ? 9 : 6.5));
        proj.vx = Math.cos(next) * spd;
        proj.vy = Math.sin(next) * spd;
      }

      proj.x += proj.vx * dt;
      proj.y += proj.vy * dt;

      if (proj.isGold && Math.random() < 0.65) {
        STATE.particles.push({
          x: proj.x, y: proj.y, vx: (Math.random() - 0.5) * 40, vy: (Math.random() - 0.5) * 40,
          life: 0.2, maxLife: 0.2, size: 3.5, color: '#fbbf24'
        });
      }

      if (proj.bounce) {
        if (proj.x < 10 || proj.x > STATE.width - 10) proj.vx *= -1;
        if (proj.y < 60 || proj.y > STATE.height - 10) proj.vy *= -1;
      }

      for (const e of STATE.enemies) {
        if (e.dead || (proj.hitSet && proj.hitSet.has(e))) continue;
        if (Math.hypot(e.x - proj.x, e.y - proj.y) <= e.radius + proj.radius) {
          if (proj.hitSet) proj.hitSet.add(e);
          damageEnemy(e, proj.dmg, proj.crit);
          if (proj.chill) e.chillTimer = 2.6;
          sfx.play('hit');

          if (proj.explode > 0) {
            STATE.shockwaves.push({ x: proj.x, y: proj.y, radius: 8, maxRadius: proj.explode, color: proj.color, alpha: 0.85 });
            for (const other of STATE.enemies) {
              if (!other.dead && other !== e && Math.hypot(other.x - proj.x, other.y - proj.y) <= proj.explode + other.radius) {
                damageEnemy(other, proj.dmg * 0.65, false);
              }
            }
          }

          proj.pierce -= 1;
          if (proj.pierce <= 0) {
            proj.life = 0;
            break;
          }
        }
      }
    }

    for (let i = STATE.enemies.length - 1; i >= 0; i--) {
      const e = STATE.enemies[i];
      if (e.dead) {
        STATE.enemies.splice(i, 1);
        continue;
      }
      if (e.hitFlash > 0) e.hitFlash -= dt;
      if (e.stunTimer > 0) {
        e.stunTimer -= dt;
        continue;
      }
      if (e.chillTimer > 0) e.chillTimer -= dt;

      const angleToPlayer = Math.atan2(p.y - e.y, p.x - e.x);
      const speedMult = e.chillTimer > 0 ? 0.55 : 1.0;
      const moveSpd = e.speed * speedMult;

      e.x += Math.cos(angleToPlayer) * moveSpd * dt;
      e.y += Math.sin(angleToPlayer) * moveSpd * dt;
      e.legPhase += dt * (moveSpd * 0.085);

      if (e.shootCd < 100) {
        e.shootTimer -= dt;
        if (e.shootTimer <= 0) {
          e.shootTimer = e.shootCd;
          const maxSwarmCap = Math.round(65 * getIntensityCfg().countMult);
          if (e.spawnsMites && STATE.enemies.length < maxSwarmCap) {
            spawnAlienByLevel(1, { x: e.x + 25, y: e.y + 20 });
            spawnAlienByLevel(1, { x: e.x - 25, y: e.y - 20 });
          } else {
            const burst = e.isBoss ? 7 : (e.alienLevel >= 9 ? 3 : 1);
            const projSpd = (e.isBoss ? 360 : 285) * getIntensityCfg().projSpeedMult;
            for (let b = 0; b < burst; b++) {
              const spread = burst > 1 ? (b - (burst - 1) / 2) * 0.22 : 0;
              const shotAng = angleToPlayer + spread;
              STATE.enemyProjectiles.push({
                x: e.x + Math.cos(shotAng) * e.radius,
                y: e.y + Math.sin(shotAng) * e.radius,
                vx: Math.cos(shotAng) * projSpd,
                vy: Math.sin(shotAng) * projSpd,
                radius: e.isBoss ? 9 : 6,
                dmg: e.dmg,
                color: e.glow,
                life: 3.4
              });
            }
          }
        }
      }

      if (Math.hypot(e.x - p.x, e.y - p.y) < e.radius + p.radius - 3) {
        handlePlayerDamage(e.dmg);
        if (STATE.mode === 'END') return;
      }
    }

    if (STATE.wave === 10) {
      const bossesAlive = STATE.enemies.some(e => e.isBoss);
      if (!bossesAlive) {
        endGame(true);
        return;
      }
    }

    for (let i = STATE.enemyProjectiles.length - 1; i >= 0; i--) {
      const ep = STATE.enemyProjectiles[i];
      ep.life -= dt;
      ep.x += ep.vx * dt;
      ep.y += ep.vy * dt;
      if (ep.life <= 0) {
        STATE.enemyProjectiles.splice(i, 1);
        continue;
      }
      if (Math.hypot(ep.x - p.x, ep.y - p.y) < ep.radius + p.radius - 3) {
        handlePlayerDamage(ep.dmg);
        STATE.enemyProjectiles.splice(i, 1);
        if (STATE.mode === 'END') return;
      }
    }

    for (let i = STATE.gems.length - 1; i >= 0; i--) {
      const g = STATE.gems[i];
      const dist = Math.hypot(p.x - g.x, p.y - g.y);
      if (dist < p.magnetRadius || g.magnetized) {
        const ang = Math.atan2(p.y - g.y, p.x - g.x);
        g.x += Math.cos(ang) * 580 * dt;
        g.y += Math.sin(ang) * 580 * dt;
      }
      if (dist < p.radius + 14) {
        p.xp += g.xp;
        p.gold += g.gold;
        sfx.play('xp');
        STATE.gems.splice(i, 1);

        if (p.xp >= p.xpToNext) {
          p.xp -= p.xpToNext;
          p.level += 1;
          p.xpToNext = Math.round(p.xpToNext * 1.24 + 18);
          sfx.play('levelup');
          updateHUD();
          openUpgradeModal(false);
          return;
        }
      }
    }

    for (let i = STATE.crates.length - 1; i >= 0; i--) {
      const c = STATE.crates[i];
      c.pulse += dt * 4;
      if (Math.hypot(p.x - c.x, p.y - c.y) < p.radius + 22) {
        STATE.crates.splice(i, 1);
        openQuadraticBreach('x² SUPPLY CACHE BREACH', false);
        return;
      }
    }

    for (let i = STATE.particles.length - 1; i >= 0; i--) {
      const pt = STATE.particles[i];
      pt.life -= dt;
      pt.x += pt.vx * dt;
      pt.y += pt.vy * dt;
      if (pt.life <= 0) STATE.particles.splice(i, 1);
    }
    for (let i = STATE.shockwaves.length - 1; i >= 0; i--) {
      const sw = STATE.shockwaves[i];
      sw.radius += (sw.maxRadius - sw.radius) * Math.min(1, dt * 11);
      sw.alpha -= dt * 2.4;
      if (sw.alpha <= 0) STATE.shockwaves.splice(i, 1);
    }
    for (let i = STATE.floatingTexts.length - 1; i >= 0; i--) {
      const ft = STATE.floatingTexts[i];
      ft.life -= dt;
      ft.y -= 34 * dt;
      if (ft.life <= 0) STATE.floatingTexts.splice(i, 1);
    }

    updateHUD();
  }

  function togglePause() {
    if (STATE.mode === 'PLAYING') {
      STATE.mode = 'PAUSED';
      document.getElementById('pauseModal').classList.remove('hidden');
    } else if (STATE.mode === 'PAUSED') {
      STATE.mode = 'PLAYING';
      document.getElementById('pauseModal').classList.add('hidden');
    }
  }

  window.addEventListener('keydown', e => {
    STATE.keys[e.code] = true;
    if (e.code === 'Space' || e.code === 'ShiftLeft' || e.code === 'ShiftRight') {
      if (STATE.mode === 'PLAYING') {
        e.preventDefault();
        triggerDash();
      }
    }
    if (e.code === 'KeyE' || e.code === 'KeyQ') {
      if (STATE.mode === 'PLAYING') {
        triggerUltimate();
      }
    }
    if (e.code === 'KeyP' || e.code === 'Escape') {
      if (STATE.mode === 'PLAYING' || STATE.mode === 'PAUSED') {
        togglePause();
      }
    }
    if (e.code === 'Enter' && STATE.mode === 'MATH') {
      resumeAfterBreach();
    }
  });

  window.addEventListener('keyup', e => {
    STATE.keys[e.code] = false;
  });

  canvas.addEventListener('mousemove', e => {
    STATE.mouse.x = e.clientX;
    STATE.mouse.y = e.clientY;
  });

  canvas.addEventListener('mousedown', e => {
    STATE.mouse.x = e.clientX;
    STATE.mouse.y = e.clientY;

    } else {
      STATE.mouse.down = true;
    }
  });

  const joyZone = document.getElementById('mobileJoystickZone');
  const joyKnob = document.getElementById('mobileJoystickKnob');

  function handleJoystickMove(clientX, clientY) {
    const rect = joyZone.getBoundingClientRect();
    const cx = rect.left + rect.width / 2;
    const cy = rect.top + rect.height / 2;
    const maxR = rect.width / 2 - 16;
    let dx = clientX - cx;
    let dy = clientY - cy;
    const dist = Math.hypot(dx, dy);
    if (dist > maxR) {
      dx = (dx / dist) * maxR;
      dy = (dy / dist) * maxR;
    }
    joyKnob.style.transform = `translate(${dx}px, ${dy}px)`;
    STATE.joystick.active = true;
    STATE.joystick.dx = dx / maxR;
    STATE.joystick.dy = dy / maxR;
  }

  joyZone.addEventListener('touchstart', e => {
    e.preventDefault();
    handleJoystickMove(e.touches[0].clientX, e.touches[0].clientY);
  }, { passive: false });

  joyZone.addEventListener('touchmove', e => {
    e.preventDefault();
    handleJoystickMove(e.touches[0].clientX, e.touches[0].clientY);
  }, { passive: false });

  joyZone.addEventListener('touchend', () => {
    STATE.joystick.active = false;
    STATE.joystick.dx = 0;
    STATE.joystick.dy = 0;
    joyKnob.style.transform = 'translate(0px, 0px)';
  });

  let lastTimestamp = 0;
  function gameLoop(timestamp) {
    if (!lastTimestamp) lastTimestamp = timestamp;
    const dt = Math.min(0.05, (timestamp - lastTimestamp) / 1000);
    lastTimestamp = timestamp;

    update(dt);
    render();
    requestAnimationFrame(gameLoop);
  }

  window.onload = () => {
    requestAnimationFrame(gameLoop);
  };
